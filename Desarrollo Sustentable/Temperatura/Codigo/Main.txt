#include <DHT.h>

#define DHTPIN 2
#define DHTTYPE DHT11
#define LEDPIN 7      // Pin donde conectarás el LED

DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(9600);
  dht.begin();
  pinMode(LEDPIN, OUTPUT);
  digitalWrite(LEDPIN, LOW); // LED apagado al iniciar
}

void loop() {
  delay(2000); // el DHT11 necesita al menos 2 seg entre lecturas

  float humedad = dht.readHumidity();
  float temperatura = dht.readTemperature(); // en Celsius por defecto

  if (isnan(humedad) || isnan(temperatura)) {
    Serial.println("Error al leer el sensor DHT11");
    return;
  }

  Serial.print("Humedad: ");
  Serial.print(humedad);
  Serial.print(" % | Temperatura: ");
  Serial.print(temperatura);
  Serial.println(" *C");

  // Control del LED según la temperatura
  if (temperatura > 26) {
    digitalWrite(LEDPIN, HIGH); // Enciende el LED
    Serial.println("LED ENCENDIDO (temp > 26°C)");
  } else {
    digitalWrite(LEDPIN, LOW);  // Apaga el LED
    Serial.println("LED APAGADO (temp <= 26°C)");
  }
}

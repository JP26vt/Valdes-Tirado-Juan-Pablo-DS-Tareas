##Arduino IDE
const int PIN_DEMANDA = A0;
const int PIN_OFERTA = A1;
const int PIN_LED = 13;
const unsigned long ESPERA_ALERTA = 60000; // maximo 1 alerta por minuto

bool enEscasez = false;
unsigned long ultimaAlerta = 0;
bool primeraAlerta = true;

void setup() {
  Serial.begin(9600);
  pinMode(PIN_LED, OUTPUT);
  mostrar("OK");
}

void loop() {
  int demanda = map(analogRead(PIN_DEMANDA), 0, 1023, 10, 90);
  int oferta = map(analogRead(PIN_OFERTA), 0, 1023, 10, 90);

  // Modelo lineal: demanda P = a - Q ; oferta P = Q - b
  int a = 100 + demanda;
  int b = oferta - 50;
  int precio = ((a - b) / 2) * 4; // pesos por kilo

  Serial.print("Demanda: "); Serial.print(demanda);
  Serial.print(" Oferta: "); Serial.print(oferta);
  Serial.print(" Precio: $"); Serial.println(precio);
  mostrar("$" + String(precio));

  bool escasez = (demanda - oferta) >= 25;
  digitalWrite(PIN_LED, escasez ? HIGH : LOW);

  if (escasez && !enEscasez &&
      (primeraAlerta || millis() - ultimaAlerta > ESPERA_ALERTA)) {
    enviarAMake(demanda, oferta, precio);
    ultimaAlerta = millis();
    primeraAlerta = false;
  }
  enEscasez = escasez;
  delay(500);
}

// La R3 no tiene matriz de LEDs: el texto sale por el Monitor Serial
void mostrar(String texto) {
  Serial.print("[Pantalla] ");
  Serial.println(texto);
}

// Entrega la alerta por USB; el script de Python la manda a Make
void enviarAMake(int demanda, int oferta, int precio) {
  Serial.print("ALERTA,");
  Serial.print(demanda); Serial.print(",");
  Serial.print(oferta); Serial.print(",");
  Serial.println(precio);
}

import serial
import requests

# ====== DATOS QUE CADA EQUIPO DEBE CAMBIAR ======
PUERTO = "COM3"   # el que ves en Arduino IDE: Herramientas > Puerto
URL_MAKE = "https://hook.us2.make.com/7soc1hcifehdxuthjgytxqza12qrvrp4"
# ================================================

arduino = serial.Serial(PUERTO, 9600, timeout=1)
print("Escuchando al Arduino... (Ctrl+C para salir)")

while True:
    linea = arduino.readline().decode("utf-8", errors="ignore").strip()
    if not linea:
        continue
    print(linea)

    if linea.startswith("ALERTA,"):
        _, demanda, oferta, precio = linea.split(",")
        r = requests.get(URL_MAKE, params={
            "demanda": demanda,
            "oferta": oferta,
            "precio": precio,
            "estado": "escasez",
        }, timeout=10)
        print("Alerta enviada a Make, respuesta:", r.status_code)
